contract TestToken {

    uint8 public decimals = 18;
    uint256 public totalSupply;
    mapping(address => uint256) public balanceOf;

    constructor(uint256 _initialSupply) {
        totalSupply = _initialSupply * 10 ** uint256(decimals);
        balanceOf[msg.sender] = totalSupply;
    }

    function transfer(address _to, uint256 _amount) public returns (bool) {
        require(balanceOf[msg.sender] >= _amount, "Insufficient balance");
        balanceOf[msg.sender] -= _amount;
        balanceOf[_to] += _amount;
        return true;
    }
}

// SPDX-License-Identifier: MIT

contract SlotMachine {
    uint256 public balance;
    event Played(address player, uint256 result);

    function deposit() public payable {
        balance += msg.value;
    }

    function play() public payable returns (uint256) {
        require(msg.value > 0, "Send ETH to play");
        uint256 random = uint256(
            keccak256(abi.encodePacked(block.timestamp, msg.sender))
        ) % 10;
        emit Played(msg.sender, random);
        if (random == 7) {
            payable(msg.sender).transfer(msg.value * 2);
        }
        return random;
    }
}

Practical No. 7 — Simulate a Blockchain Consensus Mechanism Using a Smart Contract

Consensus.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract ConsensusSimulation {
    address[] public validators;
    mapping(address => bool) public hasVoted;
    mapping(address => bool) public voteChoice;
    uint public approveCount;
    uint public rejectCount;
    bool public consensusReached;

    constructor() {
        validators.push(0x8589070a85Af9ca171d85F8210103ea1b3Ff1820); // Account 1
        validators.push(0xC502315EeaD844cc9e2E9897527607E3B2101e03); // Account 2
    }

    function vote(bool approve) public {
        require(isValidator(msg.sender), "Not a validator");
        require(!hasVoted[msg.sender], "Already voted");
        hasVoted[msg.sender] = true;
        voteChoice[msg.sender] = approve;
        if (approve) {
            approveCount++;
        } else {
            rejectCount++;
        }
        checkConsensus();
    }

    function isValidator(address user) public view returns (bool) {
        for (uint i = 0; i < validators.length; i++) {
            if (validators[i] == user) {
                return true;
            }
        }
        return false;
    }

    function checkConsensus() internal {
        if (approveCount > validators.length / 2) {
            consensusReached = true;
        }
    }

    function getVote(address user) public view returns (string memory) {
        if (!hasVoted[user]) {
            return "Not Voted";
        }
        if (voteChoice[user]) {
            return "Approved";
        } else {
            return "Rejected";
        }
    }
}

With 2 validators, consensus requires >1 approval (i.e., both must approve)
If both vote true → consensusReached = true

Auction.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract Auction {
    address public owner;
    uint public auctionEndTime;
    address public highestBidder;
    uint public highestBid;
    mapping(address => uint) public pendingReturns;
    bool public ended;

    constructor(uint _biddingTime) {
        owner = msg.sender;
        auctionEndTime = block.timestamp + _biddingTime;
    }

    modifier onlyOwner() {
        require(msg.sender == owner, "Only owner allowed");
        _;
    }

    modifier auctionActive() {
        require(block.timestamp < auctionEndTime, "Auction already ended");
        _;
    }

    modifier auctionEnded() {
        require(block.timestamp >= auctionEndTime, "Auction not yet ended");
        _;
    }

    function bid() public payable auctionActive {
        require(msg.value > highestBid, "There already is a higher bid");
        if (highestBid != 0) {
            pendingReturns[highestBidder] += highestBid;
        }
        highestBidder = msg.sender;
        highestBid = msg.value;
    }

function withdraw() public {
        uint amount = pendingReturns[msg.sender];
        require(amount > 0, "No funds to withdraw");
        
        pendingReturns[msg.sender] = 0;
        
        (bool success, ) = payable(msg.sender).call{value: amount}("");
        require(success, "Withdrawal failed");
    }

    function endAuction() public auctionEnded {
        require(!ended, "Auction already ended");
        
        ended = true;
        
        (bool success, ) = payable(owner).call{value: highestBid}("");
        require(success, "Transfer to owner failed");
    }

    function getTimeLeft() public view returns (uint) {
        if (block.timestamp >= auctionEndTime) {
            return 0;
        } else {
            return auctionEndTime - block.timestamp;
        }
    }
}

Practical No. 9 — Design a Use-Case for Blockchain in Supply Chain

SupplyChain.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract SupplyChain {
    enum State { Created, Shipped, Delivered }

    struct Product {
        uint id;
        string name;
        address manufacturer;
        State state;
    }

    mapping(uint => Product) public products;

    function createProduct(uint _id, string memory _name) public {
        products[_id] = Product(_id, _name, msg.sender, State.Created);
    }

    function shipProduct(uint _id) public {
        require(products[_id].state == State.Created, "Not in Created state");
        products[_id].state = State.Shipped;
    }

    function deliverProduct(uint _id) public {
        require(products[_id].state == State.Shipped, "Not in Shipped state");
        products[_id].state = State.Delivered;
    }

    function getProduct(uint _id) public view returns (uint, string memory, address, State) {
        Product memory p = products[_id];
        return (p.id, p.name, p.manufacturer, p.state);
    }
}

VIVA QUESTION - 

1. Why is blockchain useful in supply chain?
Blockchain provides end-to-end traceability. In traditional systems, data is siloed in private databases, making it easy to forge records. Blockchain allows verification of a product's entire journey from manufacturing to delivery, ensuring authenticity and reducing fraud.

2. What are smart contracts in finance?
Smart contracts are self-executing programs with agreement terms written directly into code. In finance, they automate processes like loan approvals, insurance payouts, and cross-border payments, eliminating manual invoicing and third-party escrow.

3. What is decentralization?
Decentralization refers to distributing authority and data across a peer-to-peer network. Every node in the blockchain network maintains a copy of the ledger, ensuring no single entity can change rules or shut down the system unilaterally.

4. What are limitations of blockchain?
Scalability: Public blockchains can be slow and expensive for high transaction volumes
Complexity: Integrating legacy systems requires specialized technical knowledge
Privacy: Transaction details on public ledgers may be visible to all participants

1. What is a DApp?
A DApp (Decentralized Application) is an application that operates on a blockchain or peer-to-peer network rather than a single centralized server. Its backend logic is governed by smart contracts, ensuring transparency, security, and resistance to censorship.

2. What is the role of MetaMask?
MetaMask serves as a bridge (provider) that allows a browser to interact with the Ethereum blockchain. It provides identity management (account addresses) and transaction signing (approving transactions before sending to the network).

3. Difference between Web2 and Web3?
Web2: Data is owned and managed by centralized companies on private servers.
Web3: Data is stored on decentralized blockchains where users have full ownership and interact through trustless smart contracts.
4. What is gas in Ethereum?
Gas is a unit measuring the computational effort required to execute operations on Ethereum. Every transaction requires a fee paid in ETH to compensate validators for processing and securing the transaction.

Viva Questions
1. What is a consensus mechanism?
A consensus mechanism is a process used in blockchain to ensure all nodes agree on the same data or ledger state. It maintains trust without a central authority and helps validate transactions and prevent fraud.

2. What is the role of validators?
Validators are participants who verify transactions and vote or propose new blocks in the blockchain. They ensure that only valid data is added to the ledger. Their role is crucial for network security.

3. What is majority consensus?
Majority consensus means more than 50% of validators agree on a decision or block. Once the majority approves, the transaction is considered valid. It ensures fairness and prevents a single participant from controlling the system.

4. What happens if consensus is not reached?
If consensus is not reached, the block or transaction is rejected or not added to the blockchain. The network waits for agreement or retries the process. This prevents incorrect or malicious data from entering the system.

5. Difference between PoW and PoS?
Proof of Work (PoW): Requires solving complex mathematical problems (used by Bitcoin). High energy consumption.
Proof of Stake (PoS): Selects validators based on their cryptocurrency stake (used by Ethereum). More energy-efficient and faster.
Viva Questions
1. What is a blockchain explorer?
A blockchain explorer is a web tool used to view transactions, blocks, and wallet details on a blockchain network. It provides transparency by allowing anyone to track activity on the network.

2. What is a block in blockchain?
A block is a container that stores a group of verified transactions. Each block is linked to the previous one using a cryptographic hash, forming a chain. It ensures data integrity and security.

3. What is a transaction hash?
A transaction hash (Tx Hash) is a unique identifier for every transaction on the blockchain. It is generated using cryptographic algorithms and helps users track and verify transaction details.

4. What information does a block contain?
A block contains transaction data, a timestamp, a nonce, the hash of the previous block, and its own unique hash. This structure ensures the chain remains secure and tamper-proof.

5. What is gas fee?
A gas fee is the cost required to perform a transaction or execute a smart contract on a blockchain. It is paid to network validators for processing work and is measured in gwei (1 ETH = 10^9 gwei).

6. Why are testnets used?
Testnets are separate blockchain networks used for testing without real money. Developers use them to test smart contracts and applications safely before deploying on the main network.
